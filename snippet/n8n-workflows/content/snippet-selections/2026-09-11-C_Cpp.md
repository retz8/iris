bool cancel(task_id_t task_id) {
  std::lock_guard lg(_task_mutex);

  auto it = _timer_tasks.begin();
  for (; it < _timer_tasks.cend(); ++it) {
    const __task &task = std::get<1>(*it);

    if (&*task == task_id) {
      _timer_tasks.erase(it);

      return true;
    }
  }

  return false;
}
