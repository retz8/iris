void SmoothBidir(float* v, size_t n, float tau_ms) {
  if (n == 0) return;
  const float alpha = std::exp(-kFrameMs / tau_ms);
  for (size_t i = 1; i < n; ++i) {
    v[i] = alpha * v[i - 1] + (1.0f - alpha) * v[i];
  }
  for (size_t i = n - 1; i-- > 0;) {
    v[i] = alpha * v[i + 1] + (1.0f - alpha) * v[i];
  }
}
