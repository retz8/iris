function scanObject(t: string, i: number): ObjectSpan { const open = i; i = skipWs(t, i + 1); const members: Member[] = []; if (t[i] === "}") return { open, close: i, members }; for (;;) { const start = i; const keyEnd = scanString(t, i); const key = JSON.parse(t.slice(start, keyEnd)) as string; i = skipWs(t, keyEnd); if (t[i] !== ":") throw new ScanError("expected a colon"); const valueStart = skipWs(t, i + 1); const valueEnd = scanValue(t, valueStart); members.push({ key, start, valueStart, valueEnd }); i = skipWs(t, valueEnd); if (t[i] === ",") { i = skipWs(t, i + 1); continue; } if (t[i] === "}") return { open, close: i, members }; throw new ScanError("expected a comma or a closing brace"); } }



Before scrolling: what does this do?

The Breakdown

  • What it does: Walks one JSON object from its opening brace and records, for every member, where its key starts and where its value begins and ends, recursing into nested values only to skip over them, never building them.
  • Key responsibility: Plannotator writes its entry into agent config files such as ~/.cursor/mcp.json. These offsets let it add or remove exactly one member as a text edit, leaving every other byte, comment and formatting choice in the user's file untouched.
  • The clever part: Why not JSON.parse and stringify? Re-serializing would reformat the file, round big integers, turn 1.0 into 1 and drop duplicate keys. Only key strings go through JSON.parse, so escapes decode correctly while values stay raw text.
